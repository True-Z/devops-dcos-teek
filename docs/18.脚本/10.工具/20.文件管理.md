## 文件格式

### KebabCase

```js
import * as fs from 'node:fs'
import * as path from 'node:path'

/**
 * 将目录中除 node_modules 外的文件和文件夹名称改为以 - 分隔，支持自定义后缀过滤和排除目录
 * @param directoryPath - 目标目录路径
 * @param extensions - 允许处理的文件后缀数组，如 ['.ts', '.js', '.d.ts']
 * @param excludeDirs - 需要排除的目录名数组，如 ['node_modules', 'dist', 'build']
 * @param processDirectories - 是否处理目录名称，默认为 true
 */
export function renameFilesWithKebabCase(directoryPath, extensions, excludeDirs, processDirectories) {
  const items = fs.readdirSync(directoryPath)

  items.forEach((item) => {
    // 检查是否在排除目录列表中
    if (excludeDirs.includes(item)) {
      return
    }

    const oldPath = path.join(directoryPath, item)
    const stats = fs.statSync(oldPath)

    // 如果是文件且指定了后缀过滤，则检查文件后缀
    if (stats.isFile() && extensions && extensions.length > 0) {
      const fileExt = path.extname(item)
      // 如果文件有后缀，检查是否在指定后缀列表中
      if (fileExt) {
        if (!extensions.includes(fileExt)) {
          // 不在指定后缀列表中的文件跳过处理
          return
        }
      } else {
        // 如果没有后缀，尝试从文件名中提取可能的后缀
        const extractedExt = extractExtensionFromBasename(item, extensions)
        if (!extractedExt) {
          // 如果无法从文件名中提取到匹配的后缀，跳过处理
          return
        }
      }
    }

    // 根据参数决定是否处理目录名称
    let newName = item
    if (stats.isFile() || (stats.isDirectory() && processDirectories)) {
      // 将名称转换为 kebab-case，保留文件扩展名
      newName = convertToKebabCaseWithExtension(item, extensions)
    }

    const newPath = path.join(directoryPath, newName)

    // 如果名称不同，则重命名
    if (item !== newName) {
      fs.renameSync(oldPath, newPath)
      console.log(`重命名: ${item} -> ${newName}`)
    }

    // 如果是目录，递归处理
    if (stats.isDirectory()) {
      renameFilesWithKebabCase(newPath, extensions, excludeDirs, processDirectories)
    }
  })
}

/**
 * 从文件名中提取可能的后缀
 * @param basename - 文件名（不包含路径）
 * @param extensions - 可能的后缀列表
 * @returns 匹配的后缀，如果没找到则返回空字符串
 */
function extractExtensionFromBasename(basename, extensions) {
  // 遍历可能的后缀，查找是否在文件名末尾匹配
  for (const ext of extensions) {
    // 移除扩展名开头的点号
    const cleanExt = ext.startsWith('.') ? ext.slice(1) : ext
    // 检查文件名是否以该扩展名结尾（忽略大小写）
    if (basename.toLowerCase().endsWith(cleanExt.toLowerCase())) {
      // 提取后缀部分（保持原始大小写）
      const suffix = basename.slice(Math.max(0, basename.length - cleanExt.length))
      return `.${suffix}`
    }
  }
  return ''
}

/**
 * 将字符串转换为 kebab-case 格式，保留文件扩展名
 * @param name - 原始名称
 * @param extensions - 可能的后缀列表，用于在没有后缀时从文件名中提取
 * @returns kebab-case 格式的名称
 */
function convertToKebabCaseWithExtension(name, extensions) {
  let ext = path.extname(name)
  let baseName = path.basename(name, ext)

  // 如果没有后缀且提供了扩展名列表，尝试从文件名中提取后缀
  if (!ext && extensions && extensions.length > 0) {
    const extractedExt = extractExtensionFromBasename(name, extensions)
    if (extractedExt) {
      // 从文件名中提取后缀部分
      const extLength = extractedExt.length - 1 // 减1是因为ext包含点号
      baseName = name.slice(0, Math.max(0, name.length - extLength))
      ext = extractedExt
    }
  }

  // 转换基础名称为 kebab-case
  const kebabBaseName = baseName
    .replaceAll(/([a-z])([A-Z])/g, '$1-$2') // 驼峰命名转换
    .replaceAll(/[\s_]+/g, '-') // 空格和下划线替换为连字符
    .replaceAll(/-+/g, '-') // 多个连字符合并为一个
    .toLowerCase() // 转换为小写
    .replaceAll(/[^a-z0-9-]/g, '') // 移除特殊字符

  // 重新组合基础名称和扩展名
  return kebabBaseName + ext
}

// 只处理文件，不处理目录名称

// 只处理 .ts 和 .js 文件，处理目录名称
// renameFilesWithKebabCase('./nsdk', ['.ts', '.js'], undefined, true)
```
