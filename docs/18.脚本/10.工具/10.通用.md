## 日常使用

### 音乐分类

```js
const fs = require('fs')
const path = require('path')

// 配置参数
const targetFolder = process.argv[2] || './55.音乐' // 目标文件夹路径
const allowedExtensions = ['.mp3', '.flac', '.m4a', '.wav'] // 允许的文件后缀
const minCountToCreate = 2 // 出现次数阈值
const alreadyExistMap = {} // 已存在歌曲对象

// 获取目标文件夹内所有符合条件的文件
function getFilesInFolder(folder) {
  try {
    const files = fs.readdirSync(folder)
    return files.filter((file) => {
      const ext = path.extname(file).toLowerCase()
      return allowedExtensions.includes(ext)
    })
  } catch (err) {
    console.error(`无法读取文件夹  ${folder}:  ${err.message}`)
    return []
  }
}

// 提取演唱人名称并进行统计
function analyzeFilenames(files) {
  const singerCounts = {}
  const singerMap = {} // 存储原始文件名与演唱人映射关系

  files.forEach((file) => {
    const filename = path.parse(file).name // 获取不带扩展名的文件名
    const singerPart = filename.split('-')[0] // 获取第一个"-"前的部分
    const singer = singerPart.split(',')[0].trim() // 获取第一个逗号前的部分

    // 过滤Windows非法字符
    const safeSinger = singer.replace(/[\<\>\:\*\?\"\\\/]/g, '_')

    // 统计演唱人出现次数
    singerMap[safeSinger] = singerMap[safeSinger] || []
    singerMap[safeSinger].push(file)

    singerCounts[safeSinger] = (singerCounts[safeSinger] || 0) + 1
  })

  return { singerCounts, singerMap }
}

// 创建符合要求的演唱人文件夹
function createSingerFolders(folder, singerCounts) {
  const createdFolders = []

  for (const singer in singerCounts) {
    const singerFolder = path.join(folder, singer)

    // 检查文件夹是否存在，存在则逃过创建
    if (fs.existsSync(singerFolder)) {
      console.log(`文件夹已存在:  ${singerFolder}`)
      createdFolders.push(singer)
      continue
    }

    if (singerCounts[singer] >= minCountToCreate) {
      if (!fs.existsSync(singerFolder)) {
        try {
          fs.mkdirSync(singerFolder)
          console.log(`已创建文件夹:  ${singerFolder}`)
          createdFolders.push(singer)
        } catch (err) {
          console.error(`创建文件夹失败:  ${singerFolder} -  ${err.message}`)
        }
      }
    }
  }

  return createdFolders
}

// 移动文件到对应的演唱人文件夹
function moveFilesToFolders(folder, singerMap, createdFolders) {
  let movedCount = 0
  let skippedCount = 0

  // 遍历所有已创建的演唱人文件夹
  createdFolders.forEach((singer) => {
    const singerFiles = singerMap[singer]
    const singerFolder = path.join(folder, singer)

    singerFiles.forEach((file) => {
      const oldPath = path.join(folder, file)
      const NewPath = path.join(singerFolder, file)

      // 检查目标文件是否已存在
      if (fs.existsSync(NewPath)) {
        console.log(`跳过移动:  ${file} (目标文件已存在)`)
        alreadyExistMap[file] = singerFolder
        skippedCount++
        return
      }

      // 检查文件是否已经移动过
      if (fs.existsSync(oldPath) && !fs.lstatSync(oldPath).isDirectory()) {
        try {
          fs.renameSync(oldPath, NewPath)
          console.log(`已移动:  ${file} ->  ${singerFolder}`)
          movedCount++
        } catch (err) {
          console.error(`移动文件失败:  ${file} -  ${err.message}`)
        }
      }
    })
  })

  console.log(`操作完成: 移动了  ${movedCount} 个文件，跳过了  ${skippedCount} 个文件\n`)
}

// 主函数
function main() {
  // 对给定目录下的二级目录进行处理，获取给定目录下所有目录名（特定）
  const dirData = fs.readdirSync(targetFolder)
  // 遍历处理
  dirData.forEach((dirPath) => {
    const joinDirPath = path.join(targetFolder, dirPath)
    console.log(`开始处理文件夹:  ${joinDirPath}`)
    console.log(`允许的文件后缀:  ${allowedExtensions.join(', ')}`)
    console.log(`出现次数阈值:  ${minCountToCreate}\n`)
    const files = getFilesInFolder(joinDirPath)
    console.log(`找到  ${files.length} 个符合条件的文件\n`)
    const { singerCounts, singerMap } = analyzeFilenames(files)
    // console.log('演唱人出现次数统计结果:')
    // for (const singer in singerCounts) {
    //     console.log(`-  ${singer}:  ${singerCounts[singer]} 次`)
    // }
    const createdFolders = createSingerFolders(joinDirPath, singerCounts)
    moveFilesToFolders(joinDirPath, singerMap, createdFolders)
    console.log('\n')
  })
  // 输出已存在歌曲列表对象
  if (Object.getOwnPropertyNames(alreadyExistMap).length !== 0) {
    console.log('已存在歌曲汇总：')
    console.log(alreadyExistMap)
  }
}

// 执行主函数
main()
```

### 文件命名格式检查

```js
const fs = require('fs')
const path = require('path')

const targetDir = './55.音乐'
const checkFormat = '.flac'

// 创建正则表达式，匹配符合规范的指定设施文件名
// 规范要求：使用带有空格的"-"作为分隔符，如"XXX - XXX.flac"
const validNameRegex = /^.*\s-\s.*$/i

// 存储不符合规范的文件名
const invalidFiles = []

// 递归遍历目录函数
function traverseDirectory(dirPath) {
  try {
    // 读取目录内容
    const files = fs.readdirSync(dirPath)

    files.forEach((file) => {
      const filePath = path.join(dirPath, file)
      const stats = fs.statSync(filePath)

      // 如果是文件且是指定格式
      if (stats.isFile() && path.extname(filePath) === checkFormat) {
        const filename = path.basename(filePath)

        // 检测文件名是否符合规范
        if (!validNameRegex.test(filename)) {
          invalidFiles.push(filePath)
        }
      }

      // 如果是目录，递归遍历
      if (stats.isDirectory()) {
        traverseDirectory(filePath)
      }
    })
  } catch (error) {
    console.error(`遍历目录 ${dirPath} 时出错:`, error.message)
  }
}

// 主函数
function main() {
  // 开始遍历目录
  console.log('开始检查FLAC文件命名规范...\n')
  traverseDirectory(targetDir)

  // 展示结果
  console.log(`检查完成，共发现 ${invalidFiles.length} 个不符合规范的FLAC文件:\n`)
  if (invalidFiles.length > 0) {
    console.log('不符合规范的文件列表:')
    invalidFiles.forEach((file, index) => {
      console.log(`${index + 1}.${file}`)
    })
    console.log(`\n命名规范要求: 文件名中使用带有空格的"-"作为分隔符`)
  } else {
    console.log('所有FLAC文件都符合命名规范。')
  }
}

// 执行主函数
main()
```
